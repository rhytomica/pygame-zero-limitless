    char.top = cell.height
    char.left = cell.width

    char.attack = 5

    extra_hp.clear()
    extra_attack.clear()
    enemies.clear()

    while len(enemies) < 5:
        x = random.randint(3, 12) * cell.width
        y = random.randint(3, 8) * cell.height
        enemy = Actor("images/curses/curse11", topleft=(x, y))

            enemy.health = random.randint(10, 20)
            enemy.attack = random.randint(5, 10)

# Logika menang atau kalah
def victory():
    global mode, win
    if len(enemies) == 0 and char.health > 0 and mode == 'game':
        mode = 'end'
        win = 1

    if char.health <= 0 and mode == 'game':
        mode = 'end'
        win = -1

# Mouse click handling for GUI buttons
def on_mouse_down(pos):
    global mode
    if mode == "menu":
        if new_game_btn.collidepoint(pos):
            reset_game()
    elif mode == "end":
        if return_menu_btn.collidepoint(pos):
            mode = "menu"

# Update loop
def update(dt):
    global is_attacking, current_frame, frame_timer, facing, projectile_actor

    if mode == 'game':
        # Overworld Item Pickups (Iterating backward safely)
        for i in range(len(extra_hp) - 1, -1, -1):
            if char.colliderect(extra_hp[i]):
                char.health += 5
                extra_hp.pop(i)

        for i in range(len(extra_attack) - 1, -1, -1):
            if char.colliderect(extra_attack[i]):
                char.attack += 2
                extra_attack.pop(i)

        victory()

        # Handle Active Attack Animations
        if is_attacking:
            frame_timer += dt
            if frame_timer >= attack_frame_duration:
                frame_timer -= attack_frame_duration
                current_frame += 1

                if current_frame >= len(active_animation):
                    is_attacking = False
                    projectile_actor = None
                    current_frame = 0
                    if facing == 'right':
                        char.image = 'images/gojo_sprites/stand/stand_right'
                    else:
                        char.image = 'images/gojo_sprites/stand/stand_left'
                else:
                    if is_ranged and projectile_actor:
                        projectile_actor.image = active_animation[current_frame]
                    else:
                        char.image = active_animation[current_frame]

        # Movement controls (Disabled while performing an attack)
        elif keyboard.right and char.x + cell.width < WIDTH - cell.width:
            facing = 'right'
            char.x += 0.5 * cell.width

            frame_timer += dt
            if frame_timer >= walk_frame_duration:
                frame_timer -= walk_frame_duration
                current_frame = (current_frame + 1) % len(walk_right_frames)
                char.image = walk_right_frames[current_frame]

        elif keyboard.left and char.x - cell.width > cell.width:
            facing = 'left'
            char.x -= 0.5 * cell.width
            
            frame_timer += dt
            if frame_timer >= walk_frame_duration:
                frame_timer -= walk_frame_duration
                current_frame = (current_frame + 1) % len(walk_left_frames)
                char.image = walk_left_frames[current_frame]

        elif keyboard.down and char.y + cell.height < HEIGHT - cell.height * 2:
            facing = 'down'
            char.y += cell.height

        elif keyboard.up and char.y - cell.height > cell.height:
            facing = 'up'
            char.y -= cell.height

# Key Handling for Attacks
def on_key_down(key):
    global is_attacking, current_frame, frame_timer, active_animation, facing, is_ranged, projectile_actor

    if not is_attacking and mode == 'game':
        attack_type = None
        damage_multiplier = 1.0
        is_ranged = False

        # Spacebar -> Normal Melee Punch
        if key == keys.SPACE:
            attack_type = punch_right_frames if facing == 'right' else punch_left_frames
            damage_multiplier = 1.0
            hitbox = char.get_rect()
        
        # Q Key -> Red Attack (2-Tile Range)
        elif key == keys.Q:
            attack_type = red
            damage_multiplier = 2.0
            is_ranged = True
            
        # E Key -> Blue Attack (2-Tile Range)
        elif key == keys.E:
            attack_type = blue
            damage_multiplier = 1.5
            is_ranged = True

        if attack_type:
            is_attacking = True
            active_animation = attack_type
            current_frame = 0
            frame_timer = 0.0

            # Compute hitboxes and projectile positions for ranged attacks
            if is_ranged:
                target_x = char.x
                target_y = char.y
                range_distance = cell.width * 2  # 2 tiles distance

                if facing == 'right':
                    target_x += range_distance
                elif facing == 'left':
                    target_x -= range_distance
                elif facing == 'down':
                    target_y += cell.height * 2
                elif facing == 'up':
                    target_y -= cell.height * 2

                # Spawn projectile visual at the target location
                projectile_actor = Actor(attack_type[0], center=(target_x, target_y))
                hitbox = projectile_actor.get_rect()
            
            # Deal damage to enemies in the target area
            for enemy_index in range(len(enemies) - 1, -1, -1):
                enemy = enemies[enemy_index]
                if hitbox.colliderect(enemy) and enemy.health > 0:
                    enemy.health -= char.attack * damage_multiplier
                    
                    if enemy.health > 0 and char.colliderect(enemy):
                        char.health -= enemy.attack
                    elif enemy.health <= 0:
                        if enemy.bonus == 1:
                            heart = Actor('images/bonus/heart')
                            heart.pos = enemy.pos
                            extra_hp.append(heart)
                        elif enemy.bonus == 2:
                            sword = Actor('images/bonus/cursed_energy')
                            sword.pos = enemy.pos
                            extra_attack.append(sword)
                        enemies.pop(enemy_index)
